BEGIN

/*

Bhoomika - 09/17/2025 - gross_Charge column added and logic 
Anshul - 10/23/2025  - Fixing the union all issue for combined temp table to handle column mismatch issue  and dynamic gross charge calculation
*/

  -- Set default value if parameter is NULL
  IF run_rc_to_drg IS NULL THEN
    SET run_rc_to_drg = FALSE;
  END IF;

  -- Set default value for tin_npi_priority if not a valid value
  IF lower(tin_npi_priority) not in ('tin', 'npi', 'both') THEN
    SET tin_npi_priority = 'both';  -- Default to both if invalid value
  ELSE
    SET tin_npi_priority = lower(tin_npi_priority);  -- Ensure it's lowercase
  END IF;

for row in (select lower(billing_class_value) as billing_class_value from unnest(billing_class) billing_class_value where lower(billing_class_value) != 'both')
do

--- Get file & rate details
execute immediate """create or replace temp table ngtd_prc_"""||row.billing_class_value||""" as (
select distinct inn_payer_file_group_id,  mrf_payer_name, raw_payer_name, mrf_plan_name, mrf_plan_name_level_2, raw_plan_name, billing_code_type, ngtd.billing_code, billing_code_description, billing_code_category_level1, billing_code_category_level2, negotiated_rate, negotiated_type, negotiation_arrangement, inn_negotiated_rate_id, inn_ngtd_prc_id , inn_month, inn_year,last_updated_on, additional_information,cast(null as string) as additional_payer_notes, 
load_dtm
from (
select distinct inn_id, inn_negotiated_rate_id, inn_ngtd_prc_id, billing_code_type, billing_code, negotiated_rate, negotiated_type, negotiation_arrangement, additional_information
from `MRF_Payer_Analytical_"""||client_name||""".mrf_analytical_inn_ngtd_prc`
-- filter on the billing class (institutional/professional)
where billing_class in ('"""||row.billing_class_value||"""','both')
and lower(negotiated_type) not like 'mrf adjusted%per diem%'
) ngtd
join (
select distinct inn_id, inn_payer_file_group_id,  coalesce(client_mapped_payer_name, mrf_mapped_payer_name, mrf_payer_name) as mrf_payer_name, mrf_payer_name as raw_payer_name, coalesce(client_mapped_plan_name, mrf_mapped_plan_name, mrf_plan_name) as mrf_plan_name,coalesce(client_mapped_plan_name_level_2, mrf_mapped_plan_name_level_2, mrf_plan_name) as mrf_plan_name_level_2, mrf_plan_name as raw_plan_name, last_updated_on,
CASE EXTRACT(MONTH FROM last_updated_on)
    WHEN 1 THEN 'January'
    WHEN 2 THEN 'February'
    WHEN 3 THEN 'March'
    WHEN 4 THEN 'April' 
    WHEN 5 THEN 'May'
    WHEN 6 THEN 'June'
    WHEN 7 THEN 'July'
    WHEN 8 THEN 'August'
    WHEN 9 THEN 'September'
    WHEN 10 THEN 'October'
    WHEN 11 THEN 'November'
    WHEN 12 THEN 'December'
END AS inn_month,
EXTRACT(YEAR FROM last_updated_on) AS inn_year,
load_dtm,
from `MRF_Payer_Analytical_"""||client_name||""".mrf_analytical_inn`
--- get the payer/plan mapping
left join MRF_Payer_Analytical_"""||client_name||""".mrf_analytical_payer_plan_mapping
on mrf_payer_name = source_payer_name
and mrf_plan_name = source_plan_name 
and data_source = 'payer'  /* only bring payer file mappings */
) inn
using (inn_id)
left join (
select distinct billing_code, coalesce(mrf_mapped_billing_code_long_desc, billing_code_long_desc) as billing_code_description, coalesce(mrf_mapped_billing_code_category_level1, billing_code_category_level1) as billing_code_category_level1, coalesce(mrf_mapped_billing_code_category_level2, billing_code_category_level2) as billing_code_category_level2,version_open_date,version_close_date
from `MRF_Payer_Reference.mrf_rfrnc_billing_code_dtls` 
) map
on ngtd.billing_code=map.billing_code and inn.last_updated_on BETWEEN map.version_open_date AND map.version_close_date
)
""";

execute immediate """create temp table if not exists dataset_existence as (select count(*) as table_count from region-US.INFORMATION_SCHEMA.TABLES
where table_schema='MRF_Hospital_Analytical_"""||client_name||"""')""";

if row.billing_class_value='institutional' and (select table_count from dataset_existence)>0 then
execute immediate """create or replace temp table provider_ngtd_prc_"""||row.billing_class_value||""" as (
select distinct mrf_hsptl_dtl_unique_file_id,mrf_payer_name, raw_payer_name, mrf_plan_name, mrf_plan_name_level_2, raw_plan_name, billing_code_type, ngtd.billing_code, billing_code_description,billing_code_category_level1, billing_code_category_level2,negotiated_rate,stndrd_ngtd_charge_prcntg, negotiated_type, inn_month, inn_year, last_updated_on,additional_information,additional_payer_notes,gross_charge
from (
select distinct mrf_hsptl_dtl_unique_file_id, coalesce(client_mapped_payer_name, mrf_mapped_payer_name, mrf_payer_name) as mrf_payer_name, mrf_payer_name as raw_payer_name, coalesce(client_mapped_plan_name, mrf_mapped_plan_name, mrf_plan_name) as mrf_plan_name,coalesce(client_mapped_plan_name_level_2, mrf_mapped_plan_name_level_2, mrf_plan_name) as mrf_plan_name_level_2, mrf_plan_name as raw_plan_name, billing_code_type, billing_code, negotiated_rate,stndrd_ngtd_charge_prcntg, negotiated_type,additional_information,additional_payer_notes, gross_charge
from `MRF_Hospital_Analytical_"""||client_name||""".mrf_hospital_analytical_ngtd_rates`
--- get the state of the hospital file
left join (
select distinct mrf_hsptl_dtl_unique_file_id, state
from `MRF_Hospital_Analytical_"""||client_name||""".mrf_hospital_analytical`
)  header
using (mrf_hsptl_dtl_unique_file_id)
--- get payer/plan mapping
left join MRF_Payer_Analytical_"""||client_name||""".mrf_analytical_payer_plan_mapping mapping
on mrf_payer_name = source_payer_name
and mrf_plan_name = source_plan_name 
--- join on state code only for hospital files as payer files can have multiple states and we don't map payer files
and header.state = mapping.state
and data_source = 'provider'  /* only bring provider file mappings */
where not (lower(negotiated_type) = 'percentage' and stndrd_ngtd_charge_prcntg is null)
and lower(negotiated_type) not like 'mrf adjusted%per diem%'
) ngtd
join (
select distinct mrf_hsptl_dtl_unique_file_id, last_updated_on,
CASE EXTRACT(MONTH FROM last_updated_on)
    WHEN 1 THEN 'January'
    WHEN 2 THEN 'February'
    WHEN 3 THEN 'March'
    WHEN 4 THEN 'April'
    WHEN 5 THEN 'May'
    WHEN 6 THEN 'June'
    WHEN 7 THEN 'July'
    WHEN 8 THEN 'August'
    WHEN 9 THEN 'September'
    WHEN 10 THEN 'October'
    WHEN 11 THEN 'November'
    WHEN 12 THEN 'December'
END AS inn_month,
EXTRACT(YEAR FROM last_updated_on) AS inn_year
from `MRF_Hospital_Analytical_"""||client_name||""".mrf_hospital_analytical`
) inn
using (mrf_hsptl_dtl_unique_file_id)
left join (
select distinct billing_code, coalesce(mrf_mapped_billing_code_long_desc, billing_code_long_desc) as billing_code_description, coalesce(mrf_mapped_billing_code_category_level1, billing_code_category_level1) as billing_code_category_level1, coalesce(mrf_mapped_billing_code_category_level2, billing_code_category_level2) as billing_code_category_level2,version_open_date,version_close_date
from `MRF_Payer_Reference.mrf_rfrnc_billing_code_dtls` 
) map
on ngtd.billing_code=map.billing_code and inn.last_updated_on BETWEEN map.version_open_date AND map.version_close_date
)
""";
end if;

if row.billing_class_value='institutional' and ((select table_count from dataset_existence)>0) then
execute immediate """create or replace temp table joined_base_data_"""||row.billing_class_value||""" 
cluster by negotiated_type, billing_code_type
as (
--- Get hospital details
with providers as (
  select distinct bus_nm as npi_name, corporate_parent_name as system_name, zip5_code, state_code, CITY_NAME,geo_address_latitude_number,geo_address_logitude_number, cbsa_title as msa_cbsa_description, COT_FCLT_TYP_DESC, COT_SPCL_DESC, licensed_bed_count_range, fips_cnty_desc, cot_clas_desc, inn_negotiated_rate_id, cbsa_code AS msa_code, org_ccn_id,
  --- set the rank based on the tin_npi_priority parameter
--- if the parameter is 'tin' then set rank1 for extraction type tin  and rank 2 for npi
--- if the parameter is 'npi' then set rank1 for extraction type npi and rank 2 for tin
--- if the parameter is 'both' then set rank1 for both extraction types 
  case '""" || tin_npi_priority || """'
  when 'tin' then case lower(extraction_type) when 'tin' then 1 else 2 end
  when 'npi' then case lower(extraction_type) when 'npi' then 1 else 2 end
  else 1 
  end as extraction_type_rank,
  from `MRF_Payer_Analytical_"""||client_name||""".mrf_client_provider_intermediate` prvdr
  where inn_negotiated_rate_id in (select distinct inn_negotiated_rate_id  from ngtd_prc_"""||row.billing_class_value||""")
  and npi_name_source = case '"""||row.billing_class_value||"""' when "institutional" then "IQVIA Hospital" when "professional" then "IQVIA Physician" end
),
--- get service code details
service_codes as (
  select distinct service_code as place_of_service_code, code_name as place_of_service_name, inn_ngtd_prc_id ,version_open_date,version_close_date
  from (
    select distinct inn_ngtd_prc_id, service_code
    from `MRF_Payer_Analytical_"""||client_name||""".mrf_analytical_inn_ngtd_prc_srvccd_dtl`
    where inn_ngtd_prc_id  in (select distinct inn_ngtd_prc_id from ngtd_prc_"""||row.billing_class_value||""")
  )
  left join (
    select distinct  code as service_code, code_name ,version_open_date,version_close_date
    from `MRF_Payer_Reference.mrf_rfrnc_service_code_dtls` 
  )
  using (service_code)
),
-- get the npi_data based on bus_nm from hospital_name_mapping table and last_updated_on between version_open_date and version_close_date

filtered_npi_dtls AS (
  SELECT DISTINCT
    mrf_hsptl_dtl_unique_file_id,
    npi_dtls.bus_nm,
    corp_par_nm,
    zip5_cd,
    st_cd,
    city_nm,
    geo_addr_lat_nbr,
    geo_addr_lon_nbr,
    cbsa_name,
    cot_fclt_typ_desc,
    cot_spcl_desc,
    tot_lic_bed_cnt,
    fips_cnty_desc,
    cot_clas_desc,
    cbsa_code,
    org_ccn_id,
    org_tax_id,
    version_open_date,
    version_close_date,
    last_updated_on,
-- Calculate the number of not null columns
CONCAT(addr_ln_1_txt, ',', city_nm, ',', st_cd, ',', zip5_cd) AS address,
  FROM MRF_Payer_Reference.mrf_rfrnc_npi_dtls npi_dtls
  join (SELECT DISTINCT mrf_hsptl_dtl_unique_file_id, bus_nm, last_updated_on FROM  MRF_Hospital_Analytical_"""||client_name||""".mrf_hospital_analytical  

        join MRF_Payer_Reference.mrf_hospital_name_mapping 
        on reporting_entity_name=mrf_hospital_name
        and hospital_address = source_file_address
        ) mapping
        on npi_dtls.bus_nm=mapping.bus_nm 
        where mapping.last_updated_on between version_open_date and version_close_date
),

-- filter the records based on address to get exact npi details of required bus_nm

neg_providers as (
  select distinct 
  a.mrf_hsptl_dtl_unique_file_id,
  a.bus_nm as npi_name,
  corp_par_nm AS system_name,
  zip5_cd AS zip5_code,
  st_cd AS state_code,
  INITCAP(city_nm) AS city_name,
  geo_addr_lat_nbr AS geo_address_latitude_number,
  geo_addr_lon_nbr AS geo_address_logitude_number,
  cbsa_name as  msa_cbsa_description,
  cot_fclt_typ_desc,
  cot_spcl_desc,
      CASE 
      WHEN COALESCE(tot_lic_bed_cnt) >= 350 THEN "350+"
      WHEN COALESCE(tot_lic_bed_cnt) >= 300 THEN "300-349"
      WHEN COALESCE(tot_lic_bed_cnt) >= 250 THEN "250-299"
      WHEN COALESCE(tot_lic_bed_cnt) >= 200 THEN "200-249"
      WHEN COALESCE(tot_lic_bed_cnt) >= 150 THEN "150-199"
      WHEN COALESCE(tot_lic_bed_cnt) >= 100 THEN "100-149"
      WHEN COALESCE(tot_lic_bed_cnt) >= 50 THEN "050-099"
      WHEN COALESCE(tot_lic_bed_cnt) >= 0 THEN "000-049"
    END AS Licensed_Bed_Count_Range,
  fips_cnty_desc,
  cot_clas_desc,
  cbsa_code as msa_code,
  org_ccn_id
  from filtered_npi_dtls a
  inner JOIN MRF_Payer_Reference.mrf_hospital_name_mapping mapping
  on a.bus_nm=mapping.bus_nm
  and lower(a.address) = lower(mapping.npi_rfrnc_address)

          
  union all

  select distinct
  a.mrf_hsptl_dtl_unique_file_id,
  a.bus_nm as npi_name,
  corp_par_nm AS system_name,
  zip5_cd AS zip5_code,
  st_cd AS state_code,
  INITCAP(city_nm) AS city_name,
  geo_addr_lat_nbr AS geo_address_latitude_number,
  geo_addr_lon_nbr AS geo_address_logitude_number,
  cbsa_name as  msa_cbsa_description,
  cot_fclt_typ_desc,
  cot_spcl_desc,
      CASE
      WHEN COALESCE(tot_lic_bed_cnt) >= 350 THEN "350+"
      WHEN COALESCE(tot_lic_bed_cnt) >= 300 THEN "300-349"
      WHEN COALESCE(tot_lic_bed_cnt) >= 250 THEN "250-299"
      WHEN COALESCE(tot_lic_bed_cnt) >= 200 THEN "200-249"
      WHEN COALESCE(tot_lic_bed_cnt) >= 150 THEN "150-199"
      WHEN COALESCE(tot_lic_bed_cnt) >= 100 THEN "100-149"
      WHEN COALESCE(tot_lic_bed_cnt) >= 50 THEN "050-099"
      WHEN COALESCE(tot_lic_bed_cnt) >= 0 THEN "000-049"
    END AS Licensed_Bed_Count_Range,
  fips_cnty_desc,
  cot_clas_desc,
  cbsa_code as msa_code,
  org_ccn_id
  from filtered_npi_dtls a
  where a.org_tax_id in (select CAST(REGEXP_REPLACE(ein, r'[^0-9]', '') AS INT64) as ein from MRF_Hospital_Analytical_"""||client_name||""".mrf_hospital_analytical)
)
--- join all CTEs to get the final base data
select distinct 
  inn_payer_file_group_id, mrf_payer_name, raw_payer_name, mrf_plan_name as Plan_Name, mrf_plan_name_level_2 as Plan_Name_level_2, raw_plan_name, inn_month, inn_year, last_updated_on,
  billing_code_type, billing_code, billing_code_description, billing_code_category_level1, billing_code_category_level2, 
  negotiation_arrangement,place_of_service_code, place_of_service_name, 
  npi_name, system_name, zip5_code as zip_code, state_code, city_name, geo_address_latitude_number,geo_address_logitude_number, msa_cbsa_description, fips_cnty_desc, cot_clas_desc, cot_fclt_typ_desc, cot_spcl_desc, Licensed_Bed_Count_Range, 0.0 as Medicare_Rate, '"""||client_name||"""' as client_name, msa_code, org_ccn_id, 'payer' as source_id,
  negotiated_type, additional_information,additional_payer_notes, 
  --- average the rates to reduce the number of rows in output
--- Keep all original percentage rates in a single array so that they can be unnested and used for normalization later
  case lower(negotiated_type)
    when "percentage" then array_agg(negotiated_rate)
    else null
  end as rate_array, 
  avg(negotiated_rate) as negotiated_rate, min(negotiated_rate) as min_negotiated_rate, max(negotiated_rate) as max_negotiated_rate,
  -- Added new column Neg_Rate_Distinct_Cnt to calculate distinct count of rates
  count(distinct negotiated_rate) as Neg_Rate_Distinct_Cnt,
  --- Added new column in provider data to get the gross charges so making it null for payer
  NULL as gross_charge
from providers t2
left join (
  service_codes t3
  right join ngtd_prc_"""||row.billing_class_value||""" t1
  on t1.inn_ngtd_prc_id=t3.inn_ngtd_prc_id and t1.last_updated_on between t3.version_open_date AND t3.version_close_date
)
using (inn_negotiated_rate_id)
group by inn_payer_file_group_id, mrf_payer_name, raw_payer_name, Plan_Name, Plan_Name_level_2, raw_plan_name, inn_month, inn_year, last_updated_on,
  billing_code_type, billing_code, billing_code_description, billing_code_category_level1, billing_code_category_level2, 
  negotiated_type, negotiation_arrangement,additional_information,additional_payer_notes, place_of_service_code, place_of_service_name, 
  npi_name, system_name, zip_code, state_code, city_name, geo_address_latitude_number, geo_address_logitude_number, msa_cbsa_description, fips_cnty_desc, cot_clas_desc, cot_fclt_typ_desc, cot_spcl_desc, Licensed_Bed_Count_Range, Medicare_Rate, client_name, msa_code, org_ccn_id, extraction_type_rank, load_dtm
--- rank the records based on extraction type rank,   and only keep the top rank record
qualify dense_rank() over (partition by npi_name, mrf_payer_name, plan_name, plan_name_level_2, billing_code_type, billing_code, negotiated_type order by extraction_type_rank, last_updated_on desc, load_dtm desc) = 1
union all

select mrf_hsptl_dtl_unique_file_id, mrf_payer_name, raw_payer_name,mrf_plan_name as Plan_Name, mrf_plan_name_level_2 as Plan_Name_level_2, raw_plan_name,inn_month, inn_year, last_updated_on,
billing_code_type, billing_code, billing_code_description, billing_code_category_level1, billing_code_category_level2,'ffs' as negotiation_arrangement,null as place_of_service_code,null as place_of_service_name,npi_name, system_name, zip5_code as zip_code, state_code, city_name, geo_address_latitude_number, geo_address_logitude_number, msa_cbsa_description, fips_cnty_desc, cot_clas_desc, cot_fclt_typ_desc, cot_spcl_desc, Licensed_Bed_Count_Range, 0.0 as Medicare_Rate, '"""||client_name||"""' as client_name, msa_code, org_ccn_id,'provider' as source_id,
negotiated_type,additional_information,additional_payer_notes,
--- average the rates to reduce the number of rows in output
--- Keep all original percentage rates in a single array so that they can be unnested and used for normalization later
case 
when lower(negotiated_type) like "%percentage%" and avg(negotiated_rate) is null then array_agg(stndrd_ngtd_charge_prcntg)
else null
end as rate_array,
case when lower(negotiated_type) like "%percentage%" and avg(negotiated_rate) is null then avg(stndrd_ngtd_charge_prcntg)
else avg(negotiated_rate) end as negotiated_rate,
case when lower(negotiated_type) like "%percentage%" and avg(negotiated_rate) is null then min(stndrd_ngtd_charge_prcntg)
else min(negotiated_rate) end as min_negotiated_rate,
case when lower(negotiated_type) like "%percentage%" and avg(negotiated_rate) is null then max(stndrd_ngtd_charge_prcntg)
else max(negotiated_rate) end as max_negotiated_rate,
-- Added new column Neg_Rate_Distinct_Cnt to calculate distinct count of rates
case when lower(negotiated_type) like "%percentage%" and avg(negotiated_rate) is null then count(distinct stndrd_ngtd_charge_prcntg)
else count(distinct negotiated_rate) end as Neg_Rate_Distinct_Cnt,
--- Added new column in provider data to get the gross charges for gross charge table.
avg(gross_charge) as gross_charge,
from neg_providers t4
join provider_ngtd_prc_"""||row.billing_class_value||""" t5
using (mrf_hsptl_dtl_unique_file_id)
group by mrf_hsptl_dtl_unique_file_id, mrf_payer_name, raw_payer_name, Plan_Name, Plan_Name_level_2,raw_plan_name, inn_month, inn_year, last_updated_on,

-- Create final gross charge details table in MRF_Payer_Analytical dataset using CTEs - separate for each billing class

cluster by npi_name, billing_code_type, billing_code
as (
-- Hospital gross charges CTE for all billing code types
with hospital_gross_charges_all as (
  select distinct 
    npi_name,
    mrf_payer_name,
    plan_name,
    billing_code_type,
    case
      when billing_code_type in ('CPT','HCPCS','CPT/HCPCS') then lpad(trim(billing_code), 5, '0')
      when billing_code_type in ('MS-DRG') then lpad(trim(billing_code), 3, '0')
      when billing_code_type in ('RC') then lpad(trim(billing_code), 4, '0')
      else trim(billing_code) 
    end as billing_code,
    avg(gross_charge) as hsptl_gross_charge
  from joined_base_data_"""||row.billing_class_value||""" 
  where billing_code_type in ('CPT','HCPCS','CDM','MS-DRG', 'RC','CPT/HCPCS')
    and gross_charge is not null
    and source_id='provider'
    and npi_name is not null
  group by 1,2,3,4,5
),
-- Client utilization gross charges CTE for all billing code types
client_gross_charges_all as (
  select distinct 
    npi_name,
    client_plan as plan_name,
    client_billing_code_type as billing_code_type,
    case
      when client_billing_code_type in ('CPT','HCPCS','CPT/HCPCS') then lpad(trim(client_billing_code), 5, '0')
      when client_billing_code_type in ('MS-DRG') then lpad(trim(client_billing_code), 3, '0')
      when client_billing_code_type in ('RC') then lpad(trim(client_billing_code), 4, '0')
      else trim(client_billing_code) 
    end as billing_code,
    ---sum(total_charges)/sum(cases) as client_gross_charge
    sum(client_expected_payment)/sum(client_healthsystem_volume) as client_gross_charge
  from `MRF_Payer_Analytical_"""||client_name||""".mrf_client_utilization`
  where client_billing_code_type in ('CPT','HCPCS','CDM','MS-DRG', 'RC','CPT/HCPCS')
    and client_billing_code is not null
    ---and total_charges is not null
    and client_expected_payment is not null
    and client_billing_class = '"""||row.billing_class_value||"""'
  group by 1,2,3,4
),
-- CMS/LDS gross charges CTE for all billing code types
lds_gross_charges_all as (
  select distinct
    trim(bus_nm) as npi_name,
    'Medicare' as plan_name,
    case 
      when billing_code_type = 'DRG' then 'MS-DRG'
      else billing_code_type
    end as billing_code_type,
    case
      when billing_code_type = 'DRG' then lpad(trim(billing_code), 3, '0')
      when billing_code_type in ('CPT','HCPCS','CPT/HCPCS') then lpad(trim(billing_code), 5, '0')
      when billing_code_type in ('RC') then lpad(trim(billing_code), 4, '0')
      else trim(billing_code) 
    end as billing_code,
    avg(charge_per_admit) as lds_gross_charge
  from `MRF_Payer_Reference.mrf_rfrnc_medicare_rate_dtls`
  where billing_code_type in ('CPT','HCPCS','DRG', 'RC','CPT/HCPCS')
    and charge_per_admit is not null
    and billing_code is not null
  group by 1,2,3,4
),
-- Master combinations CTE for all billing code types
master_combinations_all as (
  select distinct npi_name, plan_name, billing_code_type, billing_code
  from (
    select npi_name, plan_name, billing_code_type, billing_code 
    from hospital_gross_charges_all
    where billing_code_type in ('CPT','HCPCS','CDM','MS-DRG', 'RC','CPT/HCPCS')

    select npi_name, plan_name, billing_code_type, billing_code 
    from client_gross_charges_all
    where billing_code_type in ('CPT','HCPCS','CDM','MS-DRG', 'RC','CPT/HCPCS')

    where billing_code_type in ('CPT','HCPCS','MS-DRG', 'RC','CPT/HCPCS')

  where billing_code_type in ('CPT','HCPCS','CDM','MS-DRG', 'RC','CPT/HCPCS')

  mc.npi_name,
  mc.plan_name,
  mc.billing_code_type,
  mc.billing_code,
  hgc.hsptl_gross_charge,
  lgc.lds_gross_charge,
  cgc.client_gross_charge,
  -- Priority-based gross charge selection (Client > Hospital > CMS)
  case 
    when cgc.client_gross_charge is not null then cgc.client_gross_charge
    when hgc.hsptl_gross_charge is not null then hgc.hsptl_gross_charge
    when lgc.lds_gross_charge is not null then lgc.lds_gross_charge
    else null
  end as primary_gross_charge,
  -- Source indicator
  case 
    when cgc.client_gross_charge is not null then 'Client Data'
    when hgc.hsptl_gross_charge is not null then 'Hospital MRF'
    when lgc.lds_gross_charge is not null then 'CMS/LDS Data'

    when cgc.client_gross_charge is not null and hgc.hsptl_gross_charge is not null 
      then round(abs(cgc.client_gross_charge - hgc.hsptl_gross_charge) / hgc.hsptl_gross_charge * 100, 2)
    else null
  end as client_vs_hospital_variance_pct,
  case 
    when hgc.hsptl_gross_charge is not null and lgc.lds_gross_charge is not null 
      then round(abs(hgc.hsptl_gross_charge - lgc.lds_gross_charge) / lgc.lds_gross_charge * 100, 2)
    else null
  end as hospital_vs_cms_variance_pct,
  case 
    when cgc.client_gross_charge is not null and lgc.lds_gross_charge is not null 
      then round(abs(cgc.client_gross_charge - lgc.lds_gross_charge) / lgc.lds_gross_charge * 100, 2)
    else null
  end as client_vs_cms_variance_pct,
  '"""||row.billing_class_value||"""' as billing_class,
  current_datetime() as created_timestamp
from master_combinations_all mc
left join hospital_gross_charges_all hgc
  on mc.npi_name = hgc.npi_name
  and mc.plan_name = hgc.plan_name
  and mc.billing_code_type = hgc.billing_code_type
  and mc.billing_code = hgc.billing_code
left join client_gross_charges_all cgc
  on mc.npi_name = cgc.npi_name
  and mc.plan_name = cgc.plan_name
  and mc.billing_code_type = cgc.billing_code_type
  and mc.billing_code = cgc.billing_code
left join lds_gross_charges_all lgc
  on mc.npi_name = lgc.npi_name
  and mc.billing_code_type = lgc.billing_code_type
  and mc.billing_code = lgc.billing_code
where mc.billing_code_type in ('CPT/HCPCS','CPT','HCPCS','CDM','MS-DRG', 'RC')
  and (hgc.hsptl_gross_charge is not null 
       or cgc.client_gross_charge is not null 
       or lgc.lds_gross_charge is not null)
)""";

-- Convert percentage records to dollar amounts using gross charge details
execute immediate """create or replace temp table percentage_to_dollar_conversion_"""||row.billing_class_value||""" as (
with percentage_records as (
  -- Get percentage records that need conversion
  select *
  from joined_base_data_"""||row.billing_class_value||"""
  where lower(negotiated_type) like '%percentage%' 
    and lower(negotiated_type) not like '%adjusted%'
    and negotiated_rate is not null
    and source_id='payer'
    and negotiated_rate between 1 and 200  -- Filter for reasonable percentage range
),
converted_records as (
  -- Convert percentage to dollar amounts using gross charge data
  select 
    pr.* except(negotiated_rate, min_negotiated_rate, max_negotiated_rate, negotiated_type, rate_array),
    -- Dynamic negotiated_type based on gross_charge_source
    case 
      when gcd.gross_charge_source = 'Client Data' then 'MRF Adjusted Percentage (using Client Data)'
      when gcd.gross_charge_source = 'Hospital MRF' then 'MRF Adjusted Percentage (using Hospital MRF)'
      when gcd.gross_charge_source = 'CMS/LDS Data' then 'MRF Adjusted Percentage (using CMS/LDS Data)'
      else 'MRF Adjusted Percentage'
    end as negotiated_type,
    -- Convert percentage to dollar amount using primary gross charge
    case 
      when gcd.primary_gross_charge is not null and gcd.primary_gross_charge > 0
      then pr.negotiated_rate * gcd.primary_gross_charge / 100
      else null
    end as negotiated_rate,
    case 
      when gcd.primary_gross_charge is not null and gcd.primary_gross_charge > 0
      then pr.min_negotiated_rate * gcd.primary_gross_charge / 100
      else null
    end as min_negotiated_rate,
    case 
      when gcd.primary_gross_charge is not null and gcd.primary_gross_charge > 0
      then pr.max_negotiated_rate * gcd.primary_gross_charge / 100
      else null
    end as max_negotiated_rate,
    gcd.gross_charge_source,
    gcd.primary_gross_charge
  from percentage_records pr
  left join `MRF_Payer_Analytical_"""||client_name||""".gross_charge_details_"""||row.billing_class_value||"""` gcd
    on pr.npi_name = gcd.npi_name
    and pr.plan_name = gcd.plan_name
    and pr.billing_code_type = gcd.billing_code_type
    and pr.billing_code = gcd.billing_code
  where gcd.primary_gross_charge is not null 
    and gcd.primary_gross_charge > 0
    -- Only convert if we have a valid gross charge
    and gcd.gross_charge_source != 'No Data Available'
)
select * except(gross_charge_source, primary_gross_charge)
from converted_records
where negotiated_rate is not null
  and negotiated_rate > 0  -- Ensure we have valid converted rates
)""";
